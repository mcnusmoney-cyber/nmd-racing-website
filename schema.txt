generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

model Customer {
  id         String @id @default(cuid())
  name       String
  phone      String
  address    String
  motorcycle String
  plate      String

  warranties Warranty[]

  createdAt DateTime @default(now())
}

model Warranty {
  id         String @id @default(cuid())
  warrantyId String @unique

  product String
  color   String

  purchaseDate DateTime
  expiryDate   DateTime

  receiptImage String?

  status String @default("ACTIVE")

  customerId String
  customer   Customer @relation(fields: [customerId], references: [id])

  claims Claim[]

  createdAt DateTime @default(now())
}

model Claim {
  id      String @id @default(cuid())
  claimId String @unique

  issue String
  image String?

  status String @default("RECEIVED")

  warrantyId String
  warranty Warranty @relation(fields: [warrantyId], references: [id])

  createdAt DateTime @default(now())
}

model Dealer {
  id String @id @default(cuid())

  name     String
  province String
  phone    String

  status String @default("ACTIVE")

  createdAt DateTime @default(now())
}
